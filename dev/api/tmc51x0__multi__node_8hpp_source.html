153 uint32_t f_clk = ClockFreq::DEFAULT_F_CLK) noexcept

245 Result<void> ProgramSequentially(uint8_t send_delay = 2) noexcept {
246 // Count active devices to determine starting address
247 uint8_t num_devices = 0;
248 for (size_t i = 0; i < MaxDevices; ++i) {
249 if (drivers_[i].has_value()) {
250 num_devices++;
251 }
252 }
253
254 if (num_devices == 0) {
255 return Result<void>(ErrorCode::INVALID_STATE); // No devices to program
256 }
257
258 // Program devices in forward order (index 0, 1, 2, ...)
259 // but assign addresses in reverse order (254, 253, 252, ...)
260 // First chip: accessible at address 0 (NAI=GND)
261 // After each programming, next chip becomes accessible at address 0
262
263 uint8_t device_index = 0;
264 for (size_t i = 0; i < MaxDevices; ++i) {
265 if (!drivers_[i].has_value()) {
266 continue; // Skip empty slots
267 }
268
269 // Calculate target address: start from 254 and count down
270 // Device at index 0 gets address 254, index 1 gets 253, etc.
271 uint8_t target_address = 254 - device_index;
272
273 // Program device to target address
274 // The device is currently accessible at address 0 (after previous chip's
275 // NAO went LOW) or address 0 for the first chip (NAI=GND)
276 if (!drivers_[i]->uartConfig.ConfigureUartNodeAddress(target_address,
277 send_delay)) {
278 return Result<void>(
279 ErrorCode::INVALID_STATE); // Failed to program device
280 }
281
282 // ConfigureUartNodeAddress() already moved the driver to target_address.
283 // NAO does not change by itself: drive it low (OUTPUT bit 0 = 0) so the
284 // next chip's NAI goes low and that chip answers at address 0
285 // (TMC5160A datasheet section 5.4, figure 5.1; OUTPUT register 0x04).
286 if (!drivers_[i]->io.SetSdoCfg0Polarity(false)) {
287 return Result<void>(ErrorCode::COMM_ERROR);
288 }
289 comm_.DelayMs(1);
290
291 device_index++;
292 }
293
294 return Result<void>();
295 }

309 Result<void> ProgramDevice(uint8_t index, uint8_t send_delay = 2) noexcept {
310 if (index >= MaxDevices || !drivers_[index].has_value()) {
311 return Result<void>(ErrorCode::INVALID_STATE);
312 }
313
314 // Calculate target address: 254 - index (per datasheet)
315 uint8_t target_address = 254 - index;

345 Result<void> AddDevice(uint8_t index) noexcept {
346 // Validate index
347 if (index < num_onboard_devices_ || index >= MaxDevices) {
348 return Result<void>(ErrorCode::INVALID_STATE);
349 }
350
351 // Check if slot is already occupied
352 if (drivers_[index].has_value()) {
353 return Result<void>(ErrorCode::INVALID_STATE);
354 }
355
356 // Enforce sequential indexing: cannot skip indices
357 // All indices before this one (starting from num_onboard_devices) must be
358 // filled
359 for (uint8_t i = num_onboard_devices_; i < index; ++i) {
360 if (!drivers_[i].has_value()) {
361 return Result<void>(ErrorCode::INVALID_STATE); // Cannot skip indices
362 }
363 }
364
365 // Create device instance with initial address 0
366 // The actual address will be programmed via ProgramSequentially() or
367 // ProgramDevice()
368 drivers_[index].emplace(comm_, 0, 0);
369 num_active_devices_++;
370
371 return Result<void>();
372 }

389 Result<void> RemoveDevice(uint8_t index) noexcept {
390 // Cannot remove onboard devices
391 if (index < num_onboard_devices_) {
392 return Result<void>(ErrorCode::INVALID_STATE);
393 }
394
395 // Validate index
396 if (index >= MaxDevices) {
397 return Result<void>(ErrorCode::INVALID_STATE);
398 }
399
400 // Check if slot is actually active
401 if (!drivers_[index].has_value()) {
402 return Result<void>(ErrorCode::INVALID_STATE);
403 }
404
405 // Enforce sequential removal: can only remove from the end
406 // Check if there are any devices after this index
407 for (size_t i = index + 1; i < MaxDevices; ++i) {
408 if (drivers_[i].has_value()) {
409 return Result<void>(
410 ErrorCode::INVALID_STATE); // Cannot remove device in the middle of
411 // the chain
412 }
413 }
414
415 // Remove device
416 drivers_[index].reset();
417 num_active_devices_--;
418
419 return Result<void>();
420 }
