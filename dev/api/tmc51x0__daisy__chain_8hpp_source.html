129 explicit TMC51x0DaisyChain(CommType &comm, uint8_t num_onboard_devices,
130 uint32_t f_clk = ClockFreq::DEFAULT_F_CLK) noexcept

132 num_active_devices_(num_onboard_devices) {
133 (void)f_clk;

135 if (num_onboard_devices == 0 || num_onboard_devices > MaxDevices) {

140 // Create onboard TMC51x0 instances, one per device
141 // Each device has its position (0, 1, 2, ...) set in constructor
142 for (uint8_t i = 0; i < num_onboard_devices_; ++i) {

148 drivers_[i].reset();
149 }

151 // CRITICAL: Set the total chain length on the SpiCommInterface
152 // This enables proper response extraction using datasheet formula
153 // 40·(n-k+1)
154 UpdateChainLength();
155 }

210 Result<void> AddDevice(uint8_t position) noexcept {
211 // Validate position
212 if (position < num_onboard_devices_ || position >= MaxDevices) {
213 return Result<void>(ErrorCode::INVALID_VALUE);
214 }
215
216 // Check if slot is already occupied
217 if (drivers_[position].has_value()) {
218 return Result<void>(ErrorCode::INVALID_STATE);
219 }
220
221 // Enforce sequential positioning: cannot skip positions
222 // All positions before this one (starting from num_onboard_devices) must be
223 // filled
224 for (uint8_t i = num_onboard_devices_; i < position; ++i) {
225 if (!drivers_[i].has_value()) {
226 return Result<void>(
227 ErrorCode::INVALID_STATE); // Cannot skip positions in daisy chain
228 }
229 }
230
231 // Create device instance at specified position
232 drivers_[position].emplace(comm_, position);
233 num_active_devices_++;
234
235 // Update chain length on SpiCommInterface
236 UpdateChainLength();
237
238 return Result<void>();
239 }

257 Result<void> RemoveDevice(uint8_t position) noexcept {
258 // Cannot remove onboard devices
259 if (position < num_onboard_devices_) {
260 return Result<void>(ErrorCode::INVALID_VALUE);
261 }
262
263 // Validate position
264 if (position >= MaxDevices) {
265 return Result<void>(ErrorCode::INVALID_VALUE);
266 }
267
268 // Check if slot is actually active
269 if (!drivers_[position].has_value()) {
270 return Result<void>(ErrorCode::INVALID_STATE);
271 }
272
273 // Enforce sequential removal: can only remove from the end
274 // Check if there are any devices after this position
275 for (size_t i = position + 1; i < MaxDevices; ++i) {
276 if (drivers_[i].has_value()) {
277 return Result<void>(
278 ErrorCode::INVALID_STATE); // Cannot remove device in the middle of
279 // the chain
280 }
281 }
282
283 // Remove device
284 drivers_[position].reset();
285 num_active_devices_--;
286
287 // Update chain length on SpiCommInterface
288 UpdateChainLength();
289
290 return Result<void>();
291 }

342 void UpdateChainLength() noexcept {
343 // Find the highest active device position
344 uint8_t max_position = 0;
345 for (size_t i = 0; i < MaxDevices; ++i) {
346 if (drivers_[i].has_value()) {
347 max_position = static_cast<uint8_t>(i);
348 }
349 }
350
351 // Total chain length is max_position + 1 (0-indexed to 1-indexed)
352 uint8_t total_length = max_position + 1;
353 comm_.SetDaisyChainLength(total_length);
354 }
